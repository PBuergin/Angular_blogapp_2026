	src/app/login-page/login-page.ts:    const url = this.returnUrl() || '/';
	src/app/login-page/login-page.ts:    window.location.href = `${environment.api}/auth/login?returnUrl=${encodeURIComponent(url)}`;


	returnUrl wird mit encodeURIComponent() codiert. 
	Dies verhindert jedoch keinen Open Redirect. 
	Es muss geprüft werden, woher returnUrl stammt 
	und ob vor dem eigentlichen Redirect nur interne URLs zugelassen werden.

	in auth.guard.ts:
	const returnUrl = '/' + segments.map((segment) => segment.path).join('/');
	erzeugt: const returnUrl = '/blog/create';

	Frontend: returnUrl wird im Auth-Guard aus internen Angular-Routensegmenten erzeugt. 
	Kein externer Redirect erkennbar. 
	BFF-Verarbeitung der returnUrl noch prüfen.

	in auth-login.ts:
	
	function safeReturnUrl(value: string | null): string {
  		if (!value || !value.startsWith('/') || value.startsWith('//')) {
    			return '/';
  		}

  		return value;
	}

	und

	const returnUrl = safeReturnUrl(request.query.get('returnUrl'));
	
	wird verhindert, dass die url mit // (webseite) beginnt.


	Die returnUrl wird im BFF validiert. 
	safeReturnUrl() akzeptiert nur URLs, die mit / beginnen, und lehnt // explizit ab. 
	Ungültige bzw. externe URLs werden durch / ersetzt. 
	Dadurch wird an dieser Stelle ein Open Redirect verhindert.

	in auth-callback.ts
	
	return {
      		status: 302,
      		headers: {
        		...corsHeaders,
        		Location: oauthState.returnUrl || '/',
      		},
      		cookies: [sessionCookie(sealedSession), clearOAuthStateCookie()],
    	};


	URL-/Redirect-Check: Die returnUrl wird im BFF durch safeReturnUrl() validiert. 
	Nur interne Pfade, die mit / beginnen, werden akzeptiert; 
	externe URLs und // werden verworfen und durch / ersetzt. 
	Nach erfolgreichem Login verwendet der Callback die zuvor gespeicherte returnUrl für einen HTTP-302-Redirect. 
	Es ist kein Open Redirect erkennbar.
